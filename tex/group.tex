\chapter{Group Theory}

\section{Groups}

\begin{definition}[Multiplicative magma]
A \define{Multiplicative Magma} (or just \emph{Magma}) is a structure
$M=\structure{U,\star}$ extending a 1-sorted structure with a
\define{multiplication} operator $\star\colon U\times U\to U$.

We will often write $ab$ instead of $a\star b$, or freely use variants
like $a\cdot b$ or $a\bullet b$. We may also write the magma as a
subscript to the binary operator (i.e., write $a\star_{M}b$ to stress
$\star$ is the binary operator of $M$).
\end{definition}

\begin{definition}[Unital magmas]\mml{group_1:def 1}
Let $M$ be a multiplicative magma. We call $M$ \define{Unital}
when there exists an element $e\in M$ such that for all elements $h\in M$
we have $h\star e=h$ and also $e\star h=h$.
\end{definition}

\begin{definition}[Group-like magmas]\mml{group_1:def 2}
Let $M$ be a multiplicative magma. We call $M$ \define{Group-like}
when there exists an element $e\in M$ such that for all elements $h\in M$
we have
\begin{enumerate}
\item $h\star e=h$, and
\item $e\star h=h$, and
\item there exists an element $g\in M$ such that $h\star g=e$ and also
  $g\star h=e$
\end{enumerate}
\end{definition}

\begin{definition}[Associative magmas]\mml{group_1:def 3}
Let $M$ be a multiplicative magma. We call $M$ \define{Unital}
when for all $x,y,z\in M$ we have $(x\star y)\star z=x\star(y\star z)$.
\end{definition}

\begin{definition}[Group]\mml{group_1}
We see that a \define{Group} is a Group-like associative non-empty
multiplicative magma.
\end{definition}

\begin{theorem}\mml{group_1:1}
Let $M$ be a nonempty magma. Let $r$, $s$, $t$ be elements of $S$.
If $(r\star s)\star t=r\star(s\star t)$ and there exists an element
$e\in M$ such that for all $s'\in M$ we have $s'\star e=s'$ and
$e\star s'=s'$ and there exists an $s''\in M$ such that $s'\star s''=e$
and $s''\star s'=e$, then $M$ is a group.
\end{theorem}

\begin{theorem}\mml{group_1:3}
The magma $\structure{U=\RR,\star=+_{\RR}}$ is associative Group-like.
\end{theorem}

\begin{definition}\mml{group_1:def 4}
Let $G$ be a unital magma. We define the term $1_{G}$ to be the
element of $G$ such that for all $h\in G$ we have $h\star1_{G}=h$ and
$1_{G}\star h=h$.
\end{definition}

\begin{definition}[Notation for inverse of element in group]\mml{group_1:def 5}
Let $G$ be a group, let $h\in G$. We will write $h^{-1}$ for the
element of $G$ such that $h\star h^{-1}=1_{G}$ and $h^{-1}\star h=1_{G}$.
\end{definition}

\begin{theorem}\mml{group_1:5-10}
Let $G$ be a group, let $f$, $g$, $h\in G$ be elements of $G$. Then
all of the following hold:
\begin{enumerate}
\item If $h\star g=1_{G}$ and $g\star h=1_{G}$, then $g=h^{-1}$
\item If $h\star g=h\star f$ or $g\star h=f\star h$, then $g=f$
\item If $h\star g=h$ or $g\star h=h$, then $g=1_{G}$.
\item We have $1_{G}^{-1}=1_{G}$.
\item If $h^{-1}=g^{-1}$, then $h = g$
\item If $h^{-1}=1_{G}$, then $h=1_{G}$.
\end{enumerate}
\end{theorem}

\vfill\break
\begin{theorem}\mml{group_1:12-20}
Let $G$ be a group, let $f$, $g$, $h\in G$ be elements of $G$. Then
all of the following hold:
\begin{enumerate}
\item If $h\star g=1_{G}$, then $h=g^{-1}$ and $g=h^{-1}$
\item We have $h\star f=g$ iff $f=h^{-1}\star g$
\item $f\star h=g$ iff $f=g\star h^{-1}$
\item For any elements $x$, $y\in G$ there exists a $z\in G$ such that
  $x\star z=y$.
\item For any elements $x$, $y\in G$ there exists a $z\in G$ such that
  $z\star x=y$.
\item $(h\star g)^{-1}=g^{-1}\star h^{-1}$
\item $g\star h=h\star g$ iff $(g\star h)^{-1}=g^{-1}\star h^{-1}$
\item $g\star h=h\star g$ iff $g^{-1}\star h^{-1}=h^{-1}\star g^{-1}$
\item $g\star h=h\star g$ iff $g\star h^{-1}=h^{-1}\star g$
\end{enumerate}
\end{theorem}

\begin{definition}\mml{group_1:def 6}
Let $G$ be a group. We define the \define{Inverse Operator} of $G$ to
be the unary operator of $G$ denoted $\cdot_{G}^{-1}$ such that for
all elements $h\in G$ we have $\cdot_{G}^{-1}(h)=h^{-1}$.
\end{definition}

\begin{definition}[Power notation for elements of groups]\mml{group_1:def 8}
Let $G$ be a group. Let $h$ be an element of $G$.

Let $n$ be a natural number. Then we define $h^{n}$ inductively as
$h^{0}=1_{G}$ and $h^{n+1}=h\cdot h^{n}$.

Let $m$ be an integer.
We define the term $h^{m}$ to be
\begin{equation}
h^{m} = \cases{
h^{|m|} & if $m\geq0$\cr
(h^{-1})^{|m|} & if $m < 0$.\cr
}
\end{equation}
\end{definition}

\begin{theorem}[Properties of power notation]\mml{group_1:25-40}
Let $G$ be a group. Let $g$, $h$ be elements of $G$. We have the
following properties of power notation:
\begin{enumerate}
\item $g^{0}=1_{G}$;
\item $g^{1}=g$;
\item $g^{2}=g\cdot g$;
\item $g^{3}=g\cdot g\cdot g$;
\item $g^{2}=1$ iff $g^{-1}=g$;
\item for any $m\in\ZZ$, if $m\leq0$, we have $g^{m}=(g^{|m|})^{-1}$;
\item for any $m\in\ZZ$, $1_{G}^{m}=1_{G}$;
\item for any $m\in\ZZ$, if $m=-1$, then $g^{m}=g^{-1}$ (i.e., the
  power notation is compatible with the notation for the inverse operator);
\item for any $m,n\in\ZZ$, $g^{m+n}=(g^{m})\cdot(g^{n})$;
\item for any $m\in\ZZ$, $g^{m+1}=g^{m}\cdot g$ and $g^{m+1}=g\cdot g^{m}$;
\item for any $m,n\in\ZZ$, $g^{mn}=(g^{m})^{n}$;
\item for any $m\in\ZZ$, $g^{(-m)}=(g^{m})^{-1}$;
\item for any $m\in\ZZ$, $g^{(-m)}=(g^{-1})^{m}$;
\item for any $m\in\ZZ$, if $gh=hg$, then $(gh)^{m}=g^{m}h^{m}$;
\item for any $m\in\ZZ$, if $gh=hg$, then $g(h^{m})=h^{m}g$;
\end{enumerate}
\end{theorem}

\begin{definition}[Order of a group element]
Let $G$ be a group. Let $g$ be an element of $G$.


We\mml{group_1:def 10} say that $g$ is \define{Order Zero} if for all
$n\in\NN_{0}$, if $g^{n}=1_{G}$, then $n=0$.


We define the \define{Order}\mml{group_1:def 11} of $g$ to be the
natural number $\ord(g)$ such that $g^{\ord(g)}=1_{G}$ and for all
$m\in\NN$, $m\neq0$, such that $g^{m}=1_{G}$ we have $\ord(g)\leq m$.
In particular, if $g$ is of order zero, then $\ord(g)=0$.
\end{definition}

\begin{definition}[Cardinality of a finite group]\mml{group_1}
Let $G$ be a finite 1-sorted structure. We define $|G|$ to be the
natural number equal to the cardinality of its underlying set.

In particular,\mml{group_1:45} nonempty finite 1-sorted structures $G$ have $|G|\geq1$.
\end{definition}

\begin{definition}[Commutative multiplicative magmas]\mml{group_1:def 12}
Let $M$ be a multiplicative magma. We say $M$ is \define{Commutative}
if for all $x$, $y\in M$, we have $xy=yx$.
\end{definition}

\begin{remark}[Abelian groups reserved for additive notation]
We will follow \Mizar/'s conventions and reserve the phrase ``Abelian Group''
for the additive loop $\structure{U,+,0}$ which forms a commutative group.
This notion (of an Abelian group) should be understood as generalizing
a vector space by forgetting the scalar multiplication operation.

When we want to discuss groups which are commutative, we will speak of
``commutative groups''. If we wanted to discuss the classification of
finite simple groups, for example, then we would discuss ``finite
commutative groups''.
\end{remark}

\begin{definition}[Unity-preserving maps]\mml{group_1:def 13}
Let $G$ and $H$ be [unital] multiplicative magmas.
Let $f\colon G\to H$ be a function.
We say $f$ is \define{Unity-Preserving} if $f(1_{G})=1_{H}$.
\end{definition}

\section{Subgroups}

\begin{definition}[Inverting a subset of a group]\mml{group_2:def 1}
Let $G$ be a group. Let $A\subset G$ be a subset.
We define the \define{Inverse} of $A$ to be the subset $A^{-1}$ of $G$
equal to $A^{-1}:=\{g^{-1}\mid g\in A\}$.

We see that $(A^{-1})^{-1}=A$.
\end{definition}

\begin{theorem}\mml{group_2:2}
Let $G$ be a group, let $A$ be a subset of $G$, let $x\in G$.
Then $x\in A^{-1}$ if and only if there exists a $g\in A$ such that $x=g^{-1}$.
\end{theorem}

\begin{theorem}[Small examples of inverted subsets of groups]\mml{group_2:4-7}
Let $G$ be a group. Let $g$, $h$ be elements of $G$. Then:
\begin{enumerate}
\item $\{g\}^{-1}=\{g^{-1}\}$
\item $\{g,h\}^{-1}=\{g^{-1},h^{-1}\}$
\item $\{\}^{-1}=\{\}$
\item $G^{-1}=G$
\item for any subset $A\subset G$, we have $A\neq\emptyset$ iff $A^{-1}\neq\emptyset$
\end{enumerate}
\end{theorem}

\begin{definition}[Multiplying subsets of a multiplicative magma]\mml{group_2:def 2}\label{def:group:group2:def2}
Let $M$ be a nonempty multiplicative magma. Let $A$ and $B$ be subsets of $M$.
We define the multiplication of $AB$ to be the subset of $M$ equal to
$\{ab\mid a\in A,b\in B\}$.

Further, when $M$ is commutative, we see $AB=BA$.

If necessary, we may sometimes write $A\star B$ or $A\cdot B$ for the
multiplication of the sets.
\end{definition}

\begin{theorem}[Properties of multiplying subsets of magmas]\mml{group_2:9-18}
Let $M$ be a nonempty multiplicative magma.
Let $A$, $B$, $C$ be subsets of $M$. Then the following are all true:
\begin{enumerate}
\item $A\neq\emptyset$ and $B\neq\emptyset$ if and only if $AB\neq\emptyset$;
\item if $M$ is associative, then $(AB)C=A(BC)$;
\item if $M$ is a group, then $(AB)^{-1}=B^{-1}A^{-1}$;
\item $A(B\cup C)=(AB)\cup(AC)$;
\item $(A\cup B)C=(AC)\cup(BC)$;
\item $A(B\cap C)\subset(AB)\cap(AC)$;
\item $(A\cap B)C\subset(AC)\cap(BC)$;
\item $\emptyset\star A=\emptyset$ and $A\star\emptyset=\emptyset$;
\item if $M$ is a group and $A\neq\emptyset$, then $M\star A=M$ and
  $A\star M=M$.
\end{enumerate}
\end{theorem}

\begin{definition}[Multiplying a subset of a magma by an element]\mml{group_2:def 3,4}
Let $M$ be a nonempty multiplicative magma.
Let $g$ be an element of $M$.
Let $A$ be a subset of $M$.
Then we define the terms:
\begin{enumerate}
\item $gA$ to be the subset of $M$ equal to $gA := \{g\}\star A$;
\item $Ag$ to be the subset of $M$ equal to $Ag := A\star\{g\}$;
\end{enumerate}
\end{definition}

\begin{theorem}[Characterizing multiplying a subset of a magma by an element]\mml{group_2:27}
Let $M$ be a nonempty multiplicative magma.
Let $A$ be a subset of $M$.
Let $g$ be an element of $M$.
For any object $x$, we have $x\in gA$ if and only if there exists an
element $h$ of $M$ such that $x=gh$ and $h\in A$.
\end{theorem}

\begin{theorem}[Characterizing multiplying a subset of a magma by an element]\mml{group_2:28}
Let $M$ be a nonempty multiplicative magma.
Let $A$ be a subset of $M$.
Let $g$ be an element of $M$.
For any object $x$, we have $x\in Ag$ if and only if there exists an
element $h$ of $M$ such that $x=hg$ and $h\in A$.
\end{theorem}

\begin{theorem}[Associativity of multiplying subsets and elements of magmas]\mml{group_2:29-37}
Let $M$ be an associative nonempty multiplicative magma.
Let $A$, $B$ be subsets of $M$.
Let $g$, $h$ be elements of $M$. Then the following all hold:
\begin{enumerate}
\item $(gA)B=g(AB)$
\item $(Ag)B=A(gB)$
\item $(AB)g=A(Bg)$
\item $(gh)A=g(hA)$
\item $(gA)h=g(Ah)$
\item $(Ag)h=A(gh)$
\item $\emptyset\star g=\emptyset$ and $g\star\emptyset=\emptyset$
\item $Mg=M$ and $gM=M$
\item If $M$ is unital, $1_{M}A=A$ and $A=A1_{M}$.
\end{enumerate}
\end{theorem}

\begin{definition}[Subgroup]\mml{group_2:def 5}
Let $G$ be a group. We define [the soft type of] a \define{Subgroup}
of $G$ to be a group $H$ such that
\begin{enumerate}
\item the carrier of $H$ is a subset of $G$: $U_{H}\subset U_{G}$;
\item the binary operator is inherited from $G$:
  $\star_{H}=\star_{G}|_{H\times H}$.
\end{enumerate}
We will use the notation $H\subgroup G$ to indicate $H$ is a subgroup
of $G$.

This is well-defined because $G$ is a subgroup of itself.
\end{definition}

\begin{theorem}[Subgroups of finite groups are finite]\mml{group_2:39}
Let $G$ be a finite group. Let $H$ be a subgroup of $G$.
Then $H$ is finite.
\end{theorem}

\begin{theorem}[Elements of subgroup belong to parent group]\mml{group_2:40-42}
Let $G$ be a group. Let $H$ be a subgroup of $G$.
If $x$ is an element of $H$, then $x$ is an element of $G$.
This means $x\in H$ implies $x\in G$.
\end{theorem}

\begin{theorem}[Compatibility of multiplying elements of subgroup]\mml{group_2:43}
Let $G$ be a group. Let $H\subgroup G$ be a subgroup of $G$.
Let $g_{1}$, $g_{2}$ be elements of $G$.
Let $h_{1}$, $h_{2}$ be elements of $H$.
If $g_{1}=h_{1}$ and $g_{2}=h_{2}$, then $g_{1}g_{2}=h_{1}h_{2}$.
\end{theorem}

\begin{theorem}[Identity element of subgroups]\mml{group_2:44-47}
Let $G$ be a group.
Let $H\subgroup G$, $H_{1}\subgroup G$, and $H_{2}\subgroup G$ be subgroups of $G$.
Then the following all hold:
\begin{enumerate}
\item $1_{H}=1_{G}$
\item $1_{H_{1}}=1_{H_{2}}$
\item $1_{G}\in H$
\item $1_{H_{1}}\in H_{2}$
\end{enumerate}
\end{theorem}

\begin{theorem}[Inverting elements of subgroups]\mml{group_2:48}
Let $G$ be a group.
Let $H\subgroup G$ be a subgroup of $G$.
Let $h$ be an element of $H$, let $g$ be an element of $G$.
If $g=h$, then $g^{-1}=h^{-1}$.
\end{theorem}

\begin{theorem}[Multiplying elements of subgroup]\mml{group_2:50}
Let $G$ be a group. Let $g_{1}$ and $g_{2}$ be elements of $G$.
Let $H\subgroup G$ be a subgroup.
If $g_{1}\in H$ and $g_{2}\in H$, then $g_{1}g_{2}\in H$.
\end{theorem}

\begin{theorem}[Group inversion operator closed on subgroups]\mml{group_2:51}
Let $G$ be a group. Let $H\subgroup G$. Let $g$ be an element of $G$.
If $g\in H$, then $g^{-1}\in H$.
\end{theorem}

\begin{theorem}[Subgroups of commutative groups are commutative]\mml{group_2:53}
Let $G$ be a commutative group. Let $H\subgroup G$ be a subgroup.
Then $H$ is commutative.
\end{theorem}
(This is a registered fact in \texttt{GROUP\_2}.)

\begin{theorem}[Every group is a subgroup of itself]\mml{group_2:54}
Let $G$ be a group. Then $G$ is a subgroup of $G$.
\end{theorem}

\begin{theorem}[``Is subgroup'' is antisymmetric relation]\mml{group_2:55}
Let $G$ be a group.
Let $H_{1}$, $H_{2}$ be strict subgroups of $G$.
If $H_{1}$ is a subgroup of $H_{2}$, and if $H_{2}$ is a subgroup of $H_{1}$,
then $H_{1}=H_{2}$. (This justifies the $H_{1}\leq G$ notation.)
\end{theorem}

\begin{theorem}[Transitivity of subgroups]\mml{group_2:56}
Let $G_{1}$, $G_{2}$, $G_{3}$ be groups.
If $G_{1}$ is a subgroup of $G_{2}$, and $G_{2}$ is a subgroup of $G_{3}$,
then $G_{1}$ is a subgroup of $G_{3}$.
\end{theorem}

\begin{theorem}[Subsets and subgroups]\mml{group_2:57}
Let $G$ be a group.
Let $H_{1}$, $H_{2}$ be strict subgroups of $G$.
If $H_{1}\subset H_{2}$, 
then $H_{1}\subgroup H_{2}$. 
\end{theorem}


\begin{definition}[Trivial groups]\mml{group_2:def 7,8}
Let $G$ be a group. We define the \define{(proper) Trivial Group} of $G$ to be
the strict subgroup $\TrivialSubgroup{G}$ of $G$ whose set equals
$U_{\TrivialSubgroup{G}}=\{1_{G}\}$ consists of the singleton with the
identity element.

We also define the \define{(improper) Trivial Group} of $G$ to be the
strict subgroup $\Omega_{G}$ of $G$ whose underlying set is all of
$G$: $U_{\Omega_{G}}=U_{G}$.
\end{definition}

\begin{theorem}\mml{group_5:1}
Let $G$ be a group. For any object $x$, $x\in\TrivialSubgroup{G}$ iff $x=1_{G}$.
\end{theorem}

\begin{theorem}[Properties of proper trivial subgroups]\mml{group_2:63-65}
Let $G$ be a group. Let $H$, $H_{1}$, $H_{2}$ be subgroups of $G$.
Then the following all hold:
\begin{enumerate}
\item $\TrivialSubgroup{H}=\TrivialSubgroup{G}$
\item $\TrivialSubgroup{H_{1}}=\TrivialSubgroup{H_{2}}$
\item $\TrivialSubgroup{G}$ is a subgroup of $H$
\end{enumerate}
\end{theorem}

\begin{theorem}\mml{group_2:68}
For any group, $\TrivialSubgroup{G}$ is finite.
\end{theorem}

\begin{definition}[Intersection of subgroups]\mml{group_2:def 10}
Let $G$ be a group.
Let $H_{1}$ and $H_{2}$ be subgroups of $G$.
We define the term $H_{1}\cap H_{2}$ to be the strict subgroup of $G$
such that its carrier is the intersection of the underlying sets of
$H_{1}$ and $H_{2}$, i.e., $U_{H_{1}\cap H_{2}}=U_{H_{1}}\cap U_{H_{2}}$.

Obseve this is commutative $H_{1}\cap H_{2}=H_{2}\cap H_{1}$.
\end{definition}

\begin{theorem}[Associativity of intersecting subgroups]\mml{group_2:84}
Let $G$ be a group. Let $H_{1}$, $H_{2}$, $H_{3}$ be subgroups of $G$.
Then $(H_{1}\cap H_{2})\cap H_{3}=H_{1}\cap(H_{2}\cap H_{3})$.
\end{theorem}

\begin{definition}[Left cosets of a subgroup]\mml{group_2:def 15}
Let $H\subgroup G$ be a subgroup of $G$.
Then we define the set of \define{Left Cosets} of $H$ to be the family $\LeftCosets{H}{G}$
of subsets of $G$ such that for all subsets $A\subset G$, we have
$A\in\LeftCosets{H}{G}$ iff there exists an $a\in G$ such that $A=aH$.
\end{definition}

\begin{definition}[Right cosets of a subgroup]\mml{group_2:def 16}
Let $H\subgroup G$ be a subgroup of $G$.
Then we define the set of \define{Right Cosets} of $H$ to be the family $\RightCosets{H}{G}$
of subsets of $G$ such that for all subsets $A\subset G$, we have
$A\in\RightCosets{H}{G}$ iff there exists an $a\in G$ such that $A=Ha$.
\end{definition}

\begin{theorem}[Left cosets equipotent with right cosets]\mml{group_2:136}
Let $H\subgroup G$ be a subgroup.
Then $\RightCosets{H}{G}$ is equipotent with $\LeftCosets{H}{G}$.
\end{theorem}

\begin{theorem}[Cosets partition the group]\mml{group_2:137}
Let $H\subgroup G$.
Then $\Union\RightCosets{H}{G}=U_{G}$ and $\Union\LeftCosets{H}{G}=U_{G}$.
\end{theorem}

\begin{definition}[Index of a subgroup]\mml{group_2:def 17}\label{defn:group:index-of-a-subgroup}
Let $G$ be a group. Let $H\subgroup G$ be a subgroup.
We define the \define{Index} of $H$ to be the cardinal number
equal to the cardinality of the left cosets of $H$ denoted $\Index{G}{H}:=\card{\LeftCosets{H}{G}}$
\end{definition}

\begin{theorem}[Index of subgroup is a natural number when set of cosets is finite]
Let $G$ be a group, let $H\subgroup G$ be a subgroup.
If $\LeftCosets{H}{G}$ is finite, then $\Index{G}{H}$ is a natural number.
\end{theorem}

\begin{theorem}[Lagrange's theorem for groups]\mml{group_2:147}
Let $G$ be a finite group. Let $H\subgroup G$.
Then $\card{G}=\card{H}\cdot\Index{G}{H}$.
\end{theorem}

\begin{theorem}[Index of grandchild is product of indices]\mml{group_2:149}
Let $G$ be a finite group.
Let $I\subgroup G$ and $H\subgroup G$ be subgroups of $G$.
If $I\subgroup H$, then $\Index{G}{I}=\Index{G}{H}\cdot\Index{H}{I}$.
\end{theorem}

\begin{theorem}[Cauchy's theorem]\mml{group_10:11}
Let $G$ be a finite group. Let $p$ be a prime number.
If $p$ divides $\card{G}$, then there exists an element $g\in G$ of
$G$ which has order $p$.
\end{theorem}

\subsection{Normal Subgroups}

\begin{theorem}\mml{group_3:4}
Let $G$ be a group. Let $A$, $B$, $C$, $D$ be subsets of $G$.
If $A\subset B$ and $C\subset D$, then $AC\subset BD$.
\end{theorem}

\begin{theorem}\mml{group_3:5}
Let $G$ be a group. Let $a\in G$ be an element of $G$.
Let $A$, $B$ be subsets of $G$. 
If $A\subset B$, then $aA\subset aB$ and $Aa\subset Ba$.
\end{theorem}

\begin{theorem}\mml{group_3:6}
Let $G$ be a group. Let $a\in G$ be an element of $G$.
Let $H_{1}$, $H_{2}$ be subgroups of $G$. 
If $H_{1}\subgroup H_{2}$, then $aH_{1}\subgroup aH_{2}$ and $H_{1}a\subgroup H_{2}a$.
\end{theorem}

\begin{theorem}
Let $G$ be a group. Let $a\in G$ be an element of $G$.
Let $A\subset G$ be a subset of $G$. Let $H$ be a subgroup of $G$.
\begin{enumerate}
\item\mml{group_3:9} $(Aa)H=A(aH)$
\item\mml{group_3:10} $(aH)A=a(HA)$
\item\mml{group_3:11} $(AH)a=A(Ha)$
\item\mml{group_3:12} $(Ha)A=H(aA)$
\end{enumerate}
\end{theorem}

\begin{definition}[Set of all subgroups]\mml{group_3:def 1}
Let $G$ be a group. We define the term $\subgroups(G)$ to be the nonempty set of
all strict subgroups of $G$; i.e., for all objects $x$, we have
$x\in\subgroups(G)$ if and only if $x$ is a strict subgroup of $G$.
\end{definition}

\begin{theorem}[Set of subgroups is nonempty]\mml{group_3:14}
Let $G$ be a strict subgroup. Then $G\in\subgroups(G)$.
\end{theorem}

\begin{theorem}[Set of subgroups of finite group is a finite set]\mml{group_3:15}
Let $G$ be a finite group. Then $\subgroups(G)$ is a finite set.
\end{theorem}

\begin{definition}[Conjugation of element of group]\mml{group_3:def 2}
Let $G$ be a group.
Let $a$, $b$ be elements of $G$.
We define the \define{Conjugate} of $a$ by $b$ to be the element
$a^{b}$ of $G$ equal to $a^{b} := b^{-1}ab$.
\end{definition}

\begin{theorem}\mml{group_3:16}
Let $G$ be a group. Let $a$, $b$, $g$ be elements of $G$.
If $a^{g}=b^{g}$, then $a=b$.
\end{theorem}

\begin{theorem}[Conjugation involving identity element]
Let $G$ be a group. Let $a$, $b$ be elements of $G$. Then the
following hold:
\begin{enumerate}
\item\mml{group_3:17} Conjugation of identity element is identity element $1_{G}^{a}=1_{G}$;
\item\mml{group_3:18} If $a^{b}=1_{G}$, then $a=1_{G}$;
\item\mml{group_3:19} $a^{1_{G}}=a$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Properties of conjugation]
Let $G$ be a group. Let $a$, $b$, $g$, $h$ be elements of $G$.
Then the following hold:
\begin{enumerate}
\item\mml{group_3:20} $a^{a}=a$;
\item\mml{group_3:21} $a^{a^{-1}}=a$ and $(a^{-1})^{a}=a^{-1}$;
\item\mml{group_3:22} $a^{b}=a$ if and only if $ab=ba$;
\item\mml{group_3:23} $(ab)^{g}=a^{g}b^{g}$;
\item\mml{group_3:24} $(a^{g})^{h}=a^{gh}$;
\item\mml{group_3:25} $(a^{b})^{b^{-1}}=a$ and $(a^{b^{-1}})^{b}=a$;
\item\mml{group_3:26} $(a^{-1})^{b}=(a^{b})^{-1}$;
\item\mml{group_3:27} For any natural number $n\in\NN_{0}$, $(a^{n})^{b}=(a^{b})^{n}$;
\item\mml{group_3:28} For any integer $n\in\ZZ$, $(a^{n})^{b}=(a^{b})^{n}$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Conjugation in commutative group]
Let $G$ be a group. Then the following hold:
\begin{enumerate}
\item\mml{group_3:29} If $G$ is commutative, then for any $a,b\in G$, we have $a^{b}=a$;
\item\mml{group_3:30} If $a^{b}=a$ for all $a,b\in G$, then $G$ is commutative.
\end{enumerate}
\end{theorem}

\begin{definition}[Conjugating subsets of a group by subsets]\mml{group_3:def 3}
Let $G$ be a group. Let $A$ and $B$ be subsets of $G$.
We define $A^{B}$ to be the subset of $G$ equal to
\begin{equation}
A^{B}:=\{g^{h}\mid g\in A,h\in B\}.
\end{equation}
\end{definition}

\begin{theorem}\mml{group_3:31}
Let $x$ be an object, $G$ be a group. Let $A$ and $B$ be subsets of $G$.
Then $x\in A^{B}$ if and only if there exists elements $g$, $h$ of $G$
such that $x=g^{h}$ and $g\in A$ and $h\in B$.
\end{theorem}

\begin{theorem}
Let $G$ be a group. Let $A$, $B$, $C$ be subsets of $G$. Then the
following hold:
\begin{enumerate}
\item\mml{group_3:32} $A^{B}\neq\emptyset$ if and only if
  $A\neq\emptyset$ and $B\neq\emptyset$;
\item\mml{group_3:33} $A^{B}\subset (B^{-1}A)B$
\item\mml{group_3:34} $(AB)^{C}\subset A^{C}B^{C}$;
\item\mml{group_3:35} $(A^{B})^{C}=A^{BC}$;
\item\mml{group_3:36} $(A^{-1})^{B}=(A^{B})^{-1}$;
\end{enumerate}
\end{theorem}

\begin{definition}[Conjugating elements and subsets]
Let $G$ be a group. Let $A$ be a subset of $G$. Let $g$ be an element
of $G$.
\begin{enumerate}
\item\mml{group_3:def 4} We define the term $A^{g}$ to be the subset
  of $G$ equal to $A^{g} := A^{\{g\}}$;
\item\mml{group_3:def 5} We define the term $g^{A}$ to be the subset
  of $G$ equal to $g^{A} := \{g\}^{A}$.
\end{enumerate}
\end{definition}

\begin{theorem}[Properties of conjugating elements with subsets]
Let $G$ be a group. Let $a$, $b$, $g$ be elements of $G$.
Let $A$, $B$ be subsets of $G$. Then the following hold:
\begin{enumerate}
\item\mml{group_3:43} $g^{A}\subset(A^{-1}g)A$
\item\mml{group_3:44} $(A^{B})^{g}=A^{Bg}$
\item\mml{group_3:45} $(A^{g})^{B}=A^{gB}$
\item\mml{group_3:46} $(g^{A})^{B}=g^{AB}$
\item\mml{group_3:47} $(A^{a})^{b}=A^{ab}$
\item\mml{group_3:48} $(a^{A})^{b}=a^{Ab}$
\item\mml{group_3:49} $(a^{b})^{A}=a^{bA}$
\item\mml{group_3:50} $A^{g}=(g^{-1}A)g$
\item\mml{group_3:51} $(AB)^{a}\subset (A^{a})(B^{a})$
\item\mml{group_3:52} $A^{1_{G}}=A$
\item\mml{group_3:53} If $A\neq\emptyset$, then $1_{G}^{A}=\{1_{G}\}$;
\item\mml{group_3:54} $(A^{a})^{a^{-1}}=A$ and $(A^{a^{-1}})^{a}=A$.
\end{enumerate}
\end{theorem}

\begin{definition}[Conjugating a subgroup by an element]\mml{group_3:def 6}
Let $G$ be a group, let $H\subgroup G$ be a subgroup of $G$, let $g$
be an element of $G$.
We define the term $H^{a}$ to be the strict subgroup of $G$ such that
its underling set $U_{H^{a}}=U_{H}^{a}$.
\end{definition}

\begin{theorem}\mml{group_3:58}
Let $x$ be an object, $G$ be a group, $a$ be an element of $G$,
$H\subgroup G$ be a subgroup of $G$. Then $x\in H^{a}$ if and only if
there exists an element $g$ of $G$ such that $x=g^{a}$ and $g\in H$.
\end{theorem}

\begin{theorem}[Properties of conjugating a subgroup by an element]
Let $G$ be a group. Let $H$, $H_{1}$, $H_{2}$ be subgroups of $G$.
Let $a$, $b$ be elements of $G$. Then the following holds:
\begin{enumerate}
\item\mml{group_3:59} $U_{H^{a}}=(a^{-1}H)a$
\item\mml{group_3:60} $(H^{a})^{b}=H^{ab}$
\item\mml{group_3:61} $H^{1_{G}}=H$
\item\mml{group_3:62} if $H$ is a strict subgroup of $G$, the $(H^{a})^{a^{-1}}=H$
and $(H^{a^{-1}})^{a}=H$;
\item\mml{group_3:63} $(H_{1}\cap H_{2})^{a}=(H_{1}^{a})\cap(H_{2}^{a})$
\item\mml{group_3:64} $|H^{a}|=|H|$
\item\mml{group_3:65} $H$ is finite if and only if $H^{a}$ is finite
\item\mml{group_3:67} $\TrivialSubgroup{G}^{a}=\TrivialSubgroup{G}$
\item\mml{group_3:68} If $H^{a}=\TrivialSubgroup{G}$, then $H=\TrivialSubgroup{G}$;
\item\mml{group_3:69} $G^{a}=G$
\item\mml{group_3:70} If $H^{a}=G$, then $H=G$;
\item\mml{group_3:73} If $G$ is commutative, then $H^{a}=H$
\end{enumerate}
\end{theorem}

\begin{definition}[Conjugate elements form a predicate]\mml{group_3:def 7}
Let $G$ be a group. Let $a$ and $b$ be elements of $G$.
We say $a$ and $b$ \define{are conjugated} if there exists an element
$g$ of $G$ such that $a=b^{g}$.

We see that $a$ is conjugate to itself (i.e., ``are conjugate'' is reflexive).

We see that $a$ is conjugate to $b$ implies $b$ is conjugate to $a$
(i.e., ``are conjugate'' is symmetric).
\end{definition}

\begin{theorem}[Conjugate is transitive]\mml{group_3:77}
Let $G$ be a group. Let $a$, $b$, $c$ be elements of $G$.
If $a$ is conjugate to $b$, and $b$ is conjugate to $c$, then $a$ is
conjugate to $c$.
\end{theorem}

\begin{theorem}[Only identity element is conjugate to itself]\mml{group_3:78}
Let $G$ be a group. Let $a$ be an element of $G$.
If $a$ is conjugate to $1_{G}$, then $1_{G}=a$.
\end{theorem}

\begin{definition}[Conjugacy class of element]\mml{group_3:def 8}
Let $G$ be a group. Let $a$ be an element.
We define the \define{Conjugacy Class} of $a$ to be the subset of $G$
equal to $a^{U_{G}}$. We write $\Cl(a)$ for the conjugacy class of $a$.
\end{definition}

\begin{theorem}\mml{group_3:81}
Let $G$ be a group. Let $a$ and $b$ be elements of $G$.
Then $a$ is in the conjugacy class of $b$ if and only if $a$ is
conjugate to $b$.
\end{theorem}

\begin{theorem}
Let $G$ be a group. Let $a$, $b$, and $g$ be elements of $G$.
Then the following holds:
\begin{enumerate}
\item\mml{group_3:82} $a^{g}$ is in the conjugacy class of $a$
\item\mml{group_3:83} $a$ is in the conjugacy class of $a$
\item\mml{group_3:84} if $a$ is in the conjugacy class of $b$, then $b$
  is in the conjugacy class of $a$
\item\mml{group_3:85} the conjugacy class of $a$ equals the conjugacy
  class of $b$ if and only if they are not disjoint
\item\mml{group_3:86} the conjugacy class of $a$ is equal to
  $\{1_{G}\}$ if and only if $a=1_{G}$
\item\mml{group_3:87} for any subset $A$ of $G$, then $\Cl(a)A=A\Cl(a)$
\end{enumerate}
\end{theorem}

\begin{definition}[Predicate for subsets of a group being conjugate]\mml{group_3:def 9}
Let $G$ be a group. Let $A$ and $B$ be subsets of $G$.
We say $A$ and $B$ \define{Are Conjugate} (or $A$ is conjugate to $B$)
if there exists an element $g$ of $G$ such that $A=B^{g}$.

We see that $A$ is conjugate to itself since $A^{1_{G}}=A$.

We also see if $A$ is conjugate to $B$, then $B$ is conjugate to $A$
(if $A=B^{g}$, then $B=A^{g^{-1}}$).
\end{definition}

\begin{theorem}[Transitivity of subsets being conjugate]\mml{group_3:91}
Let $G$ be a group. Let $A$, $B$, $C$ be subsets of $G$.
If $A$ is conjugate to $B$ and $B$ is conjugate to $C$,
then $A$ is conjugate to $C$.
\end{theorem}

\begin{definition}[Conjugacy class of a subset]\mml{group_3:def 10}
Let $G$ be a group, let $A$ be a subset of $G$.
We define the \define{Conjugacy Class} of $A$ to be the subset-family
of $G$ equal to $\Cl(A) := \{B\subset G\mid A\mbox{ is conjugate to }B\}$.
\end{definition}

\begin{theorem}[Properties of conjugacy class for a subset]
Let $G$ be a group. Let $A$, $B$ be subsets of $G$.
Let $g$ be elements of $G$. Then the following all hold:
\begin{enumerate}
\item\mml{group_3:95} $A\in\Cl(B)$ if and only if $A$ is conjugate to $B$
\item\mml{group_3:96} $A^{g}\in\Cl(A)$
\item\mml{group_3:97} $A\in\Cl(A)$
\item\mml{group_3:98} If $A\in\Cl(B)$, then $B\in\Cl(A)$
\item\mml{group_3:99} $\Cl(A)=\Cl(B)$ if and only if $\Cl(A)$ is not
  disjoint with $\Cl(B)$
\item\mml{group_3:100} If $G$ is finite, then $\Cl(A)$ is finite
\end{enumerate}
\end{theorem}

\begin{definition}[Conjugate subgroups]\mml{group_3:def 11}
Let $G$ be a group. Let $H_{1}$ and $H_{2}$ be subgroups of $G$.
We define the predicate saying $H_{1}$ and $H_{2}$ \define{are conjugate}
to mean there exists an element $g$ of $G$ such that $\structure{U_{H_{1}},\star_{H_{1}}}=H_{2}^{g}$.

This is a reflexive and symmetric predicate.
\end{definition}

\begin{theorem}[Transitivity of subgroups beings conjugate]\mml{group_3:105}
Let $G$ be a group. Let $H_{1}$, $H_{3}$ be strict subgroups of $G$.
Let $H_{2}$ be a subgroup of $G$.
If $H_{1}$ is conjugate to $H_{2}$ and $H_{2}$ is conjugate to $H_{3}$,
then $H_{1}$ is conjugate to $H_{3}$.
\end{theorem}

\begin{definition}[Conjugacy class of subgroup]\mml{group_3:def 12}
Let $G$ be a group, let $H$ be a subgroup of $G$.
We define the \define{Conjugacy Class} of $H$ to be the subset
$\Cl(H)$ of $G$ such that for all objects $x$, we have $x\in\Cl(H)$ if
and only if there exists a strict subgroup $H_{1}$ of $G$ such that
$x=H_{1}$ and $H$ is conjugate to $H_{1}$.
\end{definition}

\begin{definition}[Normal subgroup]\mml{group_3:def 13}
Let $G$ be a group. Let $N$ be a subgroup of $G$.
We say $N$ is \define{Normal} if for each element $a$ of $G$,
$N^{a}=\structure{U_{N},\star_{N}}$.
\end{definition}

\begin{theorem}[Trivial subgroups are normal]\mml{group_3:114}
Let $G$ be a group. Then $\TrivialSubgroup{G}$ is normal and $G$ is a
normal subgroup of $G$.
\end{theorem}

\begin{theorem}[Intersection of normal subgroups is normal]\mml{group_3:115}
Let $N_{1}$, $N_{2}$ be normal subgroups of $G$. Then $N_{1}\cap N_{2}$
is a normal subgroup of $G$.
\end{theorem}

\begin{theorem}[Every subgroup of a commutative group is normal]\mml{group_3:116}
Let $G$ be a commutative group. Let $H$ be a subgroup of $G$. Then $H$
is a normal subgroup of $G$.
\end{theorem}

\begin{theorem}[Equivalent criteria for normal subgroups]
Let $G$ be a group, let $H$ be a subgroup of $G$. Then the following hold:
\begin{enumerate}
\item\mml{group_3:117} $H$ is normal iff for each $a\in G$ we have $aH=Ha$
\item\mml{group_3:118} $H$ is normal iff for each $a\in G$ we have
  $aH\subset Ha$
\item\mml{group_3:119} $H$ is normal iff for each $a\in G$ we have
  $Ha\subset aH$
\item\mml{group_3:120} $H$ is normal iff for each subset $A\subset G$
  we have $AH=HA$
\item\mml{group_3:121} $H$ is a strict normal subgroup of $G$ iff for
  each $a\in G$ we have $H\subgroup H^{a}$ is a subgroup of $H^{a}$
\item\mml{group_3:122} $H$ is a strict normal subgroup of $G$ iff for
  each $a\in G$ we have $H^{a}\subgroup H$ is a subgroup of $H$
\end{enumerate}
\end{theorem}

\begin{theorem}[Left cosets of a normal subgroup are also right cosets]\mml{group_3:127}
Let $G$ be a group. Let $N\normalsubgroup G$ be a normal subgroup of $G$.
Then the set of left cosets of $N$ in $G$ is equal to the set of right
cosets of $N$ in $G$.
\end{theorem}

\begin{definition}[Normalizer of subset]\mml{group_3:def 14}
Let $G$ be a group. Let $A$ be a subset of $G$.
We define the \define{Normalizer} of $A$ to be the strict subgroup $\normalizer{G}{A}$
of $G$ such that its underlying set equals
$U_{\normalizer{G}{A}}:=\{h\in G\mid A=A^{h}\}$.
\end{definition}

\begin{theorem}[Elements of normalizer of subset]\mml{group_3:129}
Let $G$ be a group, let $A\subset G$ be a subset of $G$, let $x$ be a set.
Then $x\in\normalizer{G}{A}$ iff there exists an element $h$ of $G$
such that $x=h$ and $A^{h}=A$.
\end{theorem}

\begin{theorem}[Index of normalizer is cardinality of conjugacy class]\mml{group_3:130}
Let $G$ be a group. Let $A\subset G$ be a subset of $G$.
Then $\card{\Cl(A)}=\Index{G}{\normalizer{G}{A}}$.
\end{theorem}

\begin{definition}[Normalizer of subgroup]\mml{group_3:def 15}
Let $G$ be a group. Let $H$ be a subgroup of $G$.
We define the \define{Normalizer} of $H$ to be the strict subgroup
$\normalizer{G}{H}$ of $G$ equal to the normalizer of the underlying
set of $H$.
\end{definition}

\begin{theorem}[Normalizer of normal subgroups is everything]\mml{group_3:137}
Let $G$ be a group. Let $H$ be a strict subgroup of $G$.
Then $H$ is a normal subgroup of $G$ if and only if $\normalizer{G}{H}=G$.
\end{theorem}

\begin{theorem}[Normalizer of trivial subgroups is everything]\mml{group_3:138,139}
Let $G$ be a group. Then $\normalizer{G}{\TrivialSubgroup{G}}=G$ and
$\normalizer{G}{G}=G$. 
\end{theorem}

\begin{definition}[Subnormal subgroups]
Let $G$ be a group. Let $H$ be a subgroup of $G$.
We say $H$ is \define{Subnormal} if there exists a finite sequence of
subgroups $H_{0}=H$, \dots, $H_{k}=G$ such that $H_{i}\normalsubgroup H_{i+1}$.
In this case, we write $H\subnormalsubgroup G$.
\end{definition}

\subsection{Generating a subgroup from a set}

\begin{definition}[Subgroup generated by a subset]\mml{group_4:def 4}
Let $G$ be a group. Let $A$ be a subset of $G$. We define the
\define{Subgroup Generated} by $A$ to be the strict subgroup $\gr{A}$
of $G$ such that $A\subset\gr{A}$ and for $H$ being a strict subgroup
of $G$ such that $A\subset H$, we have $\gr{A}\subgroup H$.
\end{definition}

\begin{theorem}
Let $G$ be a group. Let $A$, $B$ be subsets of $G$. Then
\begin{enumerate}
\item\mml{group_4:29} If $a\in A$, then $a\in\gr{A}$
\item\mml{group_4:30} $\gr{\emptyset}=\TrivialSubgroup{G}$
\item\mml{group_4:32} If $A\subset B$, then $\gr{A}$ is a subgroup of $\gr{B}$
\item\mml{group_4:33} $\gr{A\cap B}$ is a subgroup of $\gr{A}\cap\gr{B}$
\item\mml{group_4:34} $U_{\gr{A}}=\bigcap\{B\subset G\mid\exists H\subgroup G\ldotp B=U_{H}\land A\subset H\}$
\end{enumerate}
\end{theorem}

\begin{definition}[Product of a finite sequence of magma elements]\mml{group_4:def 2}
Let $M$ be a nonempty associative unital multiplicative magma.
Let $(a_{1},\dots,a_{n})$ be a finite sequence of elements of $M$.
We define the \define{Product} of $(a_{k})$ to be the element of $M$
denoted $\prod_{k}a_{k}$ defined inductively on the length of the
sequence as, for the empty product:
\begin{equation}
\prod^{0}_{i=1}a_{i}=1_{M},
\end{equation}
and when there are elements in the finite sequence:
\begin{equation}
\prod^{n+1}_{i=1}a_{i}=\left(\prod^{n}_{i=1}a_{i}\right)a_{n+1}
\end{equation}
\end{definition}

\begin{theorem}[Product of concatenation of sequence of product of products]\mml{group_4:5}
Let $M$ be a nonempty associative unital multiplicative magma.
If $(a_{k})$ and $(b_{\ell})$ are finite sequences of $M$,
and if $(c_{j})=(a_{k})\concat(b_{\ell})$ is the concatenation of these
two sequences, then
\begin{equation}
\prod_{j}c_{j}=\left(\prod_{k}a_{k}\right)\left(\prod_{\ell}b_{\ell}\right).
\end{equation}
\end{theorem}

\begin{theorem}[Inductive definition for product]\mml{group_4:6}
Let $M$ be a nonempty unital multiplicative magma.
Let $(a_{1},\dots,a_{n})$ be a finite sequence of elements of $M$.
Let $b$ be an element of $M$. Then
\begin{equation*}
\prod_{k}(a_{k})\concat\langle
b\rangle=\left(\prod_{k}a_{k}\right)\cdot b.
\end{equation*}
In particular, we have this product equal to $(((a_{1}\cdot a_{2})\cdots)a_{n})b$.
\end{theorem}

\begin{theorem}[Product of prepending an element to a sequence]\mml{group_4:7}
Let $M$ be a nonempty associative unital multiplicative magma.
Let $(a_{1},\dots,a_{n})$ be a finite sequence of elements of $M$.
Let $a$ be an element of $M$.
Then the product of the sequence $\langle a\rangle\concat(a_{k})$
prepending the sequence with $a$ is
\begin{equation*}
\prod(\langle a\rangle\concat(a_{k}))=a\cdot\prod_{k}a_{k}.
\end{equation*}
\end{theorem}

\begin{theorem}[Empty product is identity element]\mml{group_4:8}
Let $M$ be a nonempty unital multiplicative magma.
Then the empty product is the identity element
$\prod^{0}_{i}a_{i}=1_{M}$.
\end{theorem}

\begin{theorem}[Finite products]
Let $M$ be a non empty multiplicative magma.
Then the following all hold:
\begin{enumerate}
\item\mml{group_4:9} For any $a\in M$, $\prod\langle a\rangle=a$
\item\mml{group_4:10} For any $a,b\in M$, $\prod\langle a,b\rangle=ab$
\end{enumerate}
\end{theorem}

\begin{theorem}[Products do not depend on identity elements]\mml{group_4:13}
Let $G$ be a group.
Let $(a_{1},\dots,a_{n})$ be a finite sequence of elements of $G$.
Let $(b_{1},\dots,b_{m})$ be a finite sequence of elements of $G$
obtained from $(a_{i})$ by removing any entries equal to the identity
element $1_{G}$ of $G$. Then
\begin{equation*}
\prod_{i}a_{i}=\prod_{i}b_{i}.
\end{equation*}
\end{theorem}

\begin{theorem}[Inverse of product is reverse product of inverses]\mml{group_4:14}
Let $G$ be a group. Let $(a_{1},\dots,a_{n})$ and
$(b_{1},\dots,b_{n})$ be finite sequences of $G$ such that $b_{i}=a_{n-i}^{-1}$.
Then $(\prod_{i}a_{i})^{-1}=\prod_{i}b_{i}$.
\end{theorem}

\begin{definition}[Power of a finite sequence of group elements]\mml{group_4:def 3}
Let $G$ be a group.
Let $(a_{i})=(a_{1},\dots,a_{n})$ be a finite sequence of $G$,
let $(m_{i})=(m_{1},\dots,m_{n})$ be a finite sequence of integers.
We define the term $(a_{i})^{(m_{i})}$ to be the finite sequence of
$G$ equal to $(a_{1}^{m_{1}},a_{2}^{m_{2}},\dots,a_{n}^{m_{n}})$.
\end{definition}

\begin{definition}[Maximal subgroup]\mml{group_4:def 6}
Let $G$ be a group, let $H$ be a subgroup of $G$.
We say $H$ is \define{Maximal} if $H\neq G$ and for all subgroups $K$
of $G$ such that $H\neq K$ and $H$ is a subgroup of $K$ we have $K=G$.
\end{definition}

\begin{definition}[Generating element of a group]\mml{group_4:def 5}
Let $G$ be a group. Let $a$ be an element of $G$.
We say $a$ is \define{Generating} if there exists a subset $A\subset G$
such that $\gr{A}=G$ and $\gr{A\setminus\{a\}}\neq G$.

Equivalently, we say $a$ is \define{Non-generating} if for all subsets
$A\subset G$ such that $\gr{A}=G$, we have
$\gr{A\setminus\{a\}}=G$. Otherwise, $a$ is generating.
\end{definition}

\begin{theorem}\mml{group_4:36}
Let $G$ be a group. Then $1_{G}$ is non-generating.
\end{theorem}

\begin{definition}[Frattini subgroup]\mml{group_4:def 7}
Let $G$ be a group. We define the \define{Frattini Subgroup} of $G$ to
be the strict subgroup $\FrattiniSubgroup(G)$ of $G$ such that
\begin{enumerate}
\item if there exists a maximal subgroup of $G$, then
  $\FrattiniSubgroup(G)$ is the intersection of all maximal subgroups
  of $G$; or equivalently, the underlying set of $\FrattiniSubgroup(G)$ equals the intersection of
all subsets of the maximal subgroups of $G$, i.e., $\bigcap\{A\subset G\mid\exists H\subgroup G\ldotp A=U_{H}\land H\mbox{ is maximal}\}$
\item otherwise $\FrattiniSubgroup(G)=G$.
\end{enumerate}
\end{definition}

\begin{theorem}[Frattini subgroup consists of non-generating elements]\mml{group_4:42}
Let $G$ be a strict group. For each element $a\in G$,
we have $a\in\FrattiniSubgroup(G)$ iff $a$ is not a generating element
of $G$.
\end{theorem}

\begin{theorem}[Frattini subgroup contained in every maximal subgroup]\mml{group_4:40}
Let $G$ be a group, let $H\subgroup G$ be a [strict] maximal subgroup
of $G$. Then the Frattini subgroup $\FrattiniSubgroup{G}$ is a subgroup of $H$.
\end{theorem}

\begin{remark}[Frattini subgroup when there is no maximal group]
Care must be taken: when there is no maximal subgroup, the
``na\"{\i}ve'' definition would yield the underlying set being
$\bigcap\emptyset=\emptyset$. The alternate definitions for the
Frattini subgroup would yield the trivial subgroup. We want to avoid
this, which is why we have this ``piecemeal'' definition.
\end{remark}

\begin{remark}[History]
Giovanni Frattini's ``Intorno alla generazione dei gruppi di operazioni''
(\journal[Rom. Acc. L. Rend.]{Accademia dei Lincei, Rendiconti} \volume{I} no.4 (1885) 281--285, 455--457)
introduced the Frattini subgroup as the subgroup consisting of all
non-generating elements of the group. It was given this designation by
Reinhold Baer in 1952. In Section 5 of Frattini's 1885 paper, the
first occurrence of ``the Frattini argument'' may be found.
Wolfgang Gasch\"{u}tz's ``\"{U}ber die $\Phi$-Untergruppe endlicher Gruppen''
(\journal[Math. Z.]{Mathematische Zeitschrift } \volume{58} (1953)
160--170, \doi{10.1007/BF01174137}) coined the term
``Frattini argument'' for the first time. For more about this history,
see Brescia, de~Giovanni, and Trombetti~\cite{brescia2017true}.
\end{remark}

\subsection{Lattice of subgroups}

\begin{definition}[Products of subgroups]\mml{group_4:def 8}
Let $G$ be a group. Let $H_{1}$ and $H_{2}$ be subgroups of $G$.
We define $H_{1}H_{2}$ to be the subset of $G$ equal to
$U_{H_{1}}U_{H_{2}}$ the product of their underlying subsets.
\end{definition}

\begin{theorem}[Associativity of product of subgroups]
Let $G$ be a group.
Let $H_{1}$, $H_{2}$, $H_{3}$ be subgroups of $G$.
Let $A\subset G$ be a subset of $G$, let $a\in G$ be an element of $G$.
Then the following all hold:
\begin{enumerate}
\item\mml{group_4:44} $(H_{1}H_{2})H_{3}=H_{1}(H_{2}H_{3})$
\item\mml{group_4:45} $(aH_{1})H_{2}=a(H_{1}H_{2})$
\item\mml{group_4:46} $(H_{1}H_{2})a=H_{1}(H_{2}a)$
\item\mml{group_4:47} $(AH_{1})H_{2}=A(H_{1}H_{2})$
\item\mml{group_4:48} $(H_{1}H_{2})A=H_{1}(H_{2}A)$
\end{enumerate}
\end{theorem}

\begin{definition}[``Join'' of subgroups]\mml{group_4:def 9}
Let $G$ be a group.
Let $H_{1}$ and $H_{2}$ be subgroups of $G$.
Then we define the \define{Join} of $H_{1}$ with $H_{2}$ to be the
subgroup $H_{1}\join H_{2} := \gr{U_{H_{1}}\cup U_{H_{2}}}$ generated
by all the elements of $H_{1}$ and $H_{2}$. 
\end{definition}

\begin{theorem}[Join is generated by product of subgroups]\mml{group_4:50}
Let $G$ be a group.
Let $H_{1}$ and $H_{2}$ be subgroups of $G$.
Then $H_{1}\join H_{2}=\gr{H_{1}H_{2}}$.
\end{theorem}

\begin{theorem}[Underlying set of join of permutable subgroups]\mml{group_4:51}
Let $G$ be a group.
Let $H_{1}$ and $H_{2}$ be subgroups of $G$.
If $H_{1}H_{2}=H_{2}H_{1}$, then $U_{H_{1}\join H_{2}}=H_{1}H_{2}$.
\end{theorem}

\begin{theorem}[Underlying set of join of subgroups of commutative group]\mml{group_4:52}
Let $G$ be a group.
Let $H_{1}$ and $H_{2}$ be subgroups of $G$.
If $G$ is a commutative group, then $U_{H_{1}\join H_{2}}=H_{1}H_{2}$.
\end{theorem}

\begin{theorem}[Underlying set of join of normal subgroups]\mml{group_4:53}
Let $G$ be a group.
Let $N_{1}$ and $N_{2}$ be normal subgroups of $G$.
Then $U_{N_{1}\join N_{2}}=N_{1}N_{2}$.
\end{theorem}

\begin{theorem}[Join of normal subgroups is normal]\mml{group_4:54}
Let $G$ be a group.
Let $N_{1}$ and $N_{2}$ be normal subgroups of $G$.
Then $N_{1}\join N_{2}$ is a normal subgroup of $G$.
\end{theorem}

\begin{theorem}[Commutativity of join of subgroups]\mml{group_4:56}
Let $G$ be a group.
Let $H_{1}$ and $H_{2}$ be subgroups of $G$.
Then $H_{1}\join H_{2}=H_{2}\join H_{1}$.
\end{theorem}

\begin{theorem}[Associativity of join of subgroups]\mml{group_4:57}
Let $G$ be a group.
Let $H_{1}$, $H_{2}$, $H_{3}$ be subgroups of $G$.
Then $(H_{1}\join H_{2})\join H_{3}=H_{1}\join(H_{2}\join H_{3})$.
\end{theorem}

\begin{theorem}[Join with trivial subgroups]
Let $G$ be a group.
Let $H$ be a strict subgroup of $G$.
Then:
\begin{enumerate}
\item\mml{group_4:58} $TrivialSubgroup{G}\join H=H$ and $H\join\TrivialSubgroup{G}=H$
\item\mml{group_4:59} $G\join H=G$ and $H\join G=G$
\end{enumerate}
\end{theorem}

\begin{theorem}[Join contains its arguments as subgroups]\mml{group_4:60}
Let $G$ be a group.
Let $H_{1}$ and $H_{2}$ be subgroups of $G$.
Then $H_{1}$ is a subgroup of $H_{1}\join H_{2}$,
and $H_{2}$ is a subgroup of $H_{1}\join H_{2}$.
\end{theorem}

\begin{theorem}[Lattice-like behaviour of join]
Let $G$ be a group.
Let $H_{1}$, $H_{2}$, $H_{3}$ be subgroups of $G$.
Then the following all hold:
\begin{enumerate}
\item\mml{group_4:61} $H_{1}\subgroup H_{2}$ iff $H_{1}\join H_{2}=H_{2}$
\item\mml{group_4:62} If $H_{1}\subgroup H_{2}$, then $H_{1}\subgroup H_{2}\join H_{3}$
\item\mml{group_4:63} If $H_{1}\subgroup H_{3}$ and $H_{2}\subgroup H_{3}$,
  then $H_{1}\join H_{2}\subgroup H_{3}$
\item\mml{group_4:64} If $H_{1}\subgroup H_{2}$, then $H_{1}\join H_{3}$
  is a subgroup of $H_{2}\join H_{3}$
\item\mml{group_4:65} $H_{1}\cap H_{2}$ is a subgroup of $H_{1}\join H_{2}$
\item\mml{group_4:66} $(H_{1}\cap H_{2})\join H_{2}=H_{2}$
\item\mml{group_4:67} $H_{1}\cap(H_{1}\join H_{2})=H_{1}$
\item\mml{group_4:68} $H_{1}\join H_{2}=H_{2}$ iff $H_{1}\cap H_{2}=H_{1}$
\end{enumerate}
\end{theorem}

\begin{definition}[Lattice of subgroups]\mml{group_4:def 12}
Let $G$ be a group. We define the \define{Lattice of Subgroups} to be
the strict lattice structure
$$L(G):=\structure{U=\subgroups(G), \join, \cap}.$$
\end{definition}

\subsection{Center of Group, Derived Subgroup}

\begin{theorem}[Subgroups closed under conjugation]\mml{group_5:2}
Let $G$ be a group.
Let $a$, $b$ be elements of $G$.
Let $H$ be a subgroup of $G$.
If $a\in H$ and $b\in H$, then $a^{b}\in H$.
\end{theorem}

\begin{theorem}[Normal subgroup closed under conjugation]\mml{group_5:3}
Let $G$ be a group.
Let $a$, $b$ be elements of $G$.
Let $N\normalsubgroup G$ be a strict normal subgroup of $G$.
If $a\in N$, then $a^{b}\in N$.
\end{theorem}

\begin{definition}[Commutator of elements of a group]\mml{group_5:def 2}
Let $G$ be a group. Let $a$, $b$ be elements of $G$.
We define the \define{Commutator} of $a$ with $b$ to be the element of
$G$ equal to $[a,b]:=(((a^{-1}b^{-1})a)b$.
\end{definition}

\begin{theorem}[Properties of commutator of group elements]
Let $G$ be a group.
Let $a$, $b$, $c$ be elements of $G$. Then the following all hold:
\begin{enumerate}
\item\mml{group_5:19} $[1_{G},a]=1_{G}$ and $[a,1_{G}]1_{G}$
\item\mml{group_5:20} $[a,a]=1_{G}$
\item\mml{group_5:21} $[a,a^{-1}]=1_{G}$ and $[a^{-1},a]=1_{G}$
\item\mml{group_5:22} $[a,b]^{-1}=[b,a]$
\item\mml{group_5:23} $[a,b]^{c}=[a^{c},b^{c}]$
\item\mml{group_5:25} $[ab,c]=[a,c]^{b}[b,c]$
\item\mml{group_5:26} $[a,bc]=[a,c]([a,b]^{c})$
\item\mml{group_5:27} $[a^{-1},b]=[b,a]^{a^{-1}}$
\item\mml{group_5:28} $[a,b^{-1}]=[b,a]^{b^{-1}}$
\item\mml{group_5:29} $[a^{-1},b^{-1}]=[a,b]^{(ab)^{-1}}$ and $[a^{-1},b^{-1}]=[a,b]^{(ba)^{-1}}$
\item\mml{group_5:30} $[a,b^{a^{-1}}]=[b,a^{-1}]$
\item\mml{group_5:31} $[a^{b^{-1}},b]=[b^{-1},a]$
\item\mml{group_5:34} for any $n\in\ZZ$, $[a^{n},b]=a^{-n}(a^{b})^{n}$
\item\mml{group_5:35} for any $n\in\ZZ$, $[a,b^{n}]=(b^{a})^{-n}b^{n}$
\item\mml{group_5:36} $[a,b]=1_{G}$ iff $ab=ba$
\item\mml{group_5:38} for any $H\subgroup G$, if $a\in H$ and $b\in H$,
  then $[a,b]\in H$
\end{enumerate}
\end{theorem}

\begin{definition}[Commutator with three elements]\mml{group_5:def 3}
Let $G$ be a group.
Let $a$, $b$, $c$ be elements of $G$.
Then we introduce the notation $[a,b,c] := [[a,b],c]$.
\end{definition}

\begin{theorem}[Hall's identity]\mml{group_5:46}
Let $G$ be a group.
Let $a$, $b$, $c$ be elements of $G$.
Then $[a,b^{-1},c]^{b}[b,c^{-1},a]^{c}[c,a^{-1},b]^{a}=1_{G}$.
\end{theorem}

\begin{definition}[Set of all commutators from subsets of a group]\mml{group_5:def 4}
Let $G$ be a group.
Let $A$ and $B$ be subsets of $G$.
We define \define{the commutators of $A$ with $B$} to be the subset of
$G$ equal to the set of all possible commutators of their elements
$\{[a,b]\mid a\in A,b\in B\}$.
\end{definition}

\begin{definition}[Group generated from commutators of subsets]\mml{group_5:def 7}
Let $G$ be a group.
Let $A$ and $B$ be subsets of $G$.
We define the subgroup $[A,B]$ of $G$ to be equal to $\gr{\{[a,b]\mid a\in A,b\in B\}}$.
Note this is the subgroup generated by the set of all commutators of
$A$ with $B$.
\end{definition}

\begin{definition}[Group generated from commutators of subgroups]\mml{group_3:def 8}
Let $G$ be a group.
Let $H_{1}$, $H_{2}$ be subgroups of $G$.
We define $[H_{1},H_{2}]$ to be the subgroup of $G$ generated by the
set of all commutators of $H_{1}$ with $H_{2}$.
\end{definition}

\begin{definition}[Derived subgroup]\mml{group_5:def 9}
Let $G$ be a group.
We define the \define{Derived Subgroup} of $G$ to be the strict normal
subgroup $G':=[G,G]$ generated from the commutators of all its elements.

This is also denoted $[G,G]$, $G^{(1)}$, rarely $\Gamma_{2}(G)$
\end{definition}

\begin{definition}[Center of a group]\mml{group_5:def 10}
Let $G$ be a group. 
We define the \define{Center} of $G$ to be the strict subgroup $Z(G)$
of $G$ such that its underlying set $U_{Z(G)}=\{a\in G\mid\forall b\in G\ldotp ab=ba\}$ 
consists of elements which commute with everything in $G$.
\end{definition}

\begin{theorem}[Group is commutative if center is everything]\mml{group_5:82}
Let $G$ be a [strict] group.
Then $G$ is commutative if and only if $Z(G)=G$.
\end{theorem}

\section{Quotient groups and Morphisms}

\subsection{Quotient groups}

\begin{definition}[Quotient group]\mml{group_6}
Let $G$ be a group. Let $N\normalsubgroup G$ be a normal subgroup of $G$.
We define the \define{Quotient Group} of $G$ with respect to $N$ to be
the strict multiplicative magma $G/N$ equal to the set of left cosets
of $N$ equipped with set-wise multiplication (\S\ref{def:group:group2:def2}).

We see that $G/N$ is strict nonempty group-like associative --- i.e.,
it's a \emph{bona fide} strict group.
\end{definition}

\begin{theorem}\mml{group_6:21}
Let $N\normalsubgroup G$.
For any element $S\in G/N$, there exists an $a\in G$ such that $S=aN=Na$.
\end{theorem}

\begin{theorem}[Identity element of quotient group]\mml{group_6:24}
Let $N\normalsubgroup G$ be a normal subgroup of $G$.
Then $1_{G/N}=U_{N}$ the identity element of the quotient group is
just the set underlying $N$.
\end{theorem}

\begin{theorem}[Inverse operator for quotient group]\mml{group_6:25}
Let $G$ be a group. Let $N\normalsubgroup G$ be a normal subgroup of $G$.
Let $S\in G/N$ be an element of the quotient group. 
If $S=aN$, then $S^{-1}=a^{-1}N$.
\end{theorem}

\begin{theorem}[Cardinality of quotient group is index of denominator]\mml{group_6:26}
Let $G$ be a group. Let $N\normalsubgroup G$ be a normal subgroup.
Then $\card{G/N}=\Index{G}{N}$.
\end{theorem}

\begin{theorem}[Quotient group is commutative iff normal subgroup contains derived subgroup]\mml{group_6:30}
Let $G$ be a strict group.
Let $N\normalsubgroup G$ be a strict normal subgroup of $G$.
Then $G/N$ is commutative iff $G'\subgroup N$.
\end{theorem}

\begin{definition}[Casting subgroups]\mml{group_6:def 1}
Let $G$ be a group. Let $B\subgroup G$ be a subgroup.
Let $N\normalsubgroup G$ be a normal subgroup of $G$.
If $N\subgroup B$, then we introduce the notation $(N)_{B}$ for the
strict normal subgroup of $B$ equal to $\structure{U_{N},\star_{N}}$ (the
multiplicative magma underlying $N$).

So $(N)_{B}\normalsubgroup B$ equals $N$. We just use this notation to
stress that we are thinking of $(N)_{B}$ as a normal subgroup of $B$.
\end{definition}

\begin{theorem}[Subgroup correspondence for quotient group]\mml{group_6:28}
Let $G$ be a group. Let $B$ be a subgroup of $G$.
Let $M\normalsubgroup G$ be a strict normal subgroup of $G$.
If $M\subgroup B$, then $B/(M)_{B}$ is a subgroup of $G/M$.
\end{theorem}

\begin{theorem}[Normal subgroup correspondence for quotient group]\mml{group_6:29}
Let $G$ be a group. Let $B$ be a subgroup of $G$.
Let $N\normalsubgroup G$ be a strict normal subgroup of $G$.
If $N\normalsubgroup B$, then $B/(N)_{B}$ is a normal subgroup of $G/N$.
\end{theorem}

\subsection{Morphisms of groups}

\begin{definition}[Multiplicative functions]\mml{group_6:def 6}
Let $G$ and $H$ be nonempty multiplicative magmas.
Let $f\colon G\to H$ be a function.
We call $f$ \define{Multiplicative} if for every element $a$ and $b$
of $G$, $f(ab)=f(a)f(b)$.
\end{definition}

\begin{theorem}[Composition of multiplicative functions is multiplicative]\mml{group_6:39}
Let $G$, $H$, $I$ be nonempty unital multiplicative magmas.
If $f\colon G\to H$ and $g\colon H\to I$ are multiplicative,
then $g\circ f\colon G\to I$ is multiplicative.
\end{theorem}

\begin{definition}[Group Morphism]\mml{group_6}
Let $G$, $H$ be groups.
We define [the soft type of] a \define{Group Morphism} from $G$ to $H$
to be a multiplicative function from $G$ to $H$.
\end{definition}

\begin{theorem}[Group morphisms preserve identity element]\mml{group_6:31}
Let $G$, $H$ be groups.
Let $f\colon G\to H$ be a group morphism.
Then $f(1_{G})=1_{H}$.
\end{theorem}

\begin{theorem}[Group morphisms preserve inverses]\mml{group_6:32}
Let $G$, $H$ be groups.
Let $f\colon G\to H$ be a group morphism.
For any $a\in G$, we have $f(a^{-1})=(f(a))^{-1}$.
\end{theorem}

\begin{theorem}[Group morphisms preserve conjugation]\mml{group_6:33}
Let $G$, $H$ be groups.
Let $f\colon G\to H$ be a group morphism.
For any $a,b\in G$, we have $f(a^{b})=(f(a))^{f(b)}$.
\end{theorem}

\begin{theorem}[Group morphisms preserve commutators]\mml{group_6:34}
Let $G$, $H$ be groups.
Let $f\colon G\to H$ be a group morphism.
For any $a,b\in G$, we have $f([a,b])=[f(a),f(b)]$.
\end{theorem}

\begin{definition}[Constant group morphism]\mml{group_6:def 7}
Let $G_{1}$ and $G_{2}$ be groups.
We define the \define{Constant Group Morphism} (or \define{Zero Morphism})
from $G_{1}$ to $G_{2}$ to be the map sending all of $G_{1}$ to the
identity element $1_{G_{2}}$.
\end{definition}

\begin{definition}[Natural morphism to quotient group]\mml{group_6:def 8}
Let $G$ be a group.
Let $N\normalsubgroup G$ be a normal subgroup of $G$.
We define the \define{Natural Morphism} (or the \define{Quotient Epimorphism}) of $N$ to be the morphism
$\nathom{N}\colon G\to G/N$ such that for all $x\in G$, $\nathom{N}(x)=xN$ is
the left coset $xN$ of $N$.

This is also called the ``canonical morphism from $G$ to $G/N$''.
\end{definition}

\begin{definition}[Kernel of group morphism]\mml{group_6:def 9}
Let $f\colon G\to H$ be a group morphism.
We define the \define{Kernel} of $f$ to be the subgroup $\ker(f)$ of
$G$ such that its underlying set equals $U_{\ker(f)}=\{a\in G\mid f(a)=1_{H}\}$.
\end{definition}

\begin{theorem}\mml{group_6:41}
Let $f\colon G\to H$ be a group morphism.
For every element $x\in G$, we have $x\in\ker(f)$ iff $f(x)=1_{H}$.
\end{theorem}

\begin{definition}[Image of a group morphism]\mml{group_6:def 10}
Let $f\colon G\to H$ be a group morphism.
We define the \define{Image} of $f$ to be the subgroup $\Im(f)$ of $H$
whose underlying set equals $U_{\Im(f)}=f(U_{G})$.
\end{definition}

\begin{theorem}\mml{group_6:45}
Let $f\colon G\to H$ be a group morphism.
For any object $x$, we have $x\in\Im(f)$ iff there exists an $a\in G$
such that $x=f(a)$.
\end{theorem}

\begin{theorem}[Injective morphisms have trivial kernels]\mml{group_6:56}
Let $f\colon G\to H$ be a group morphism.
Then $f$ is injective iff $\ker(f)=\TrivialSubgroup{G}$.
\end{theorem}

\begin{definition}[Isomorphic groups]\mml{group_6:def 11}
Let $G$, $H$ be groups.
We say $G$ and $H$ \define{Are Isomorphic} (or $G$ is isomorphic to $H$)
if there exists a bijective group morphism $f\colon G\to H$.
This is a predicate of two groups.

We write $G\iso H$ as shorthand for $G$ and $H$ being isomorphic.

Observe: for any group $G$, $G\iso G$ is isomorphic to itself.
\end{definition}

\begin{theorem}[Symmetry of isomorphic]\mml{group_6:66}
Let $G$, $H$ be groups.
If $G$ is isomorphic to $H$, then $H$ is isomorphic to $G$.
\end{theorem}

\begin{theorem}[Transitivity of isomorphic groups]\mml{group_6:67}
Let $G_{1}$, $G_{2}$, $G_{3}$ be groups.
If $G_{1}$ is isomorphic to $G_{2}$, and if $G_{2}$ is isomorphic to $G_{3}$,
then $G_{1}$ is isomorphic to $G_{3}$.
\end{theorem}

In other words, the preceding theorems taken altogether tells us that
``is isomorphic to'' forms an equivalence relation on groups.

\begin{theorem}
Let $G$, $H$ be isomorphic groups. Then the following holds:
\begin{enumerate}
\item\mml{group_6:74} If $G$ is finite, then $H$ is finite.
\item\mml{group_6:76} If $G$ is trivial, then $H$ is trivial.
\item\mml{group_6:77} If $G$ is commutative, then $H$ is commutative.
\end{enumerate}
\end{theorem}

\subsection{Isomorphism theorems}

\begin{theorem}[First isomorphism theorem for groups]\mml{group_6:79}
Let $G$ and $H$ be groups.
Let $\varphi\colon G\to H$ be a group morphism.
Then there exists a group morphism $\psi\colon G/\ker(\varphi)\to\Im(\varphi)$
such that $\psi$ is bijective and $\varphi=\psi\circ\nathom{\ker(\varphi)}$.
\end{theorem}

\begin{theorem}[Second isomorphism theorem for groups]\mml{group_6:81}
Let $G$ be a group. Let $B\subgroup G$ be a subgroup of $G$.
Let $N\normalsubgroup G$ be a [strict] normal subgroup of $G$.
Then $(B\join N)/(N)_{B\join N}\iso B/(B\cap N)$.
\end{theorem}

\begin{theorem}[Third isomorphism theorem for groups]\mml{group_6:80}
Let $G$ be a group.
Let $N\normalsubgroup G$ be a normal subgroup of $G$.
Let $M\normalsubgroup G$ be a [strict] normal subgroup of $G$.
Let $J\normalsubgroup G/M$ be a [strict] normal subgroup of $G/M$.
If $M$ is a subgroup of $N$ and if $J=N/(M)_{N}$, then $(G/M)/J\iso G/N$.
\end{theorem}

\section{Sylow Theorems}

\begin{definition}[Cyclic group]\mml{gr_cy_1:def 5}
Let $n\in\NN$ be a [positive] natural number.
We define the \define{Cyclic Group of Order $n$} to be the finite
commutative group
$\CyclicGroup{n}$ consisting of numbers $\{0,1,\dots,n-1\}$ equipped
with addition modulo $n$.
\end{definition}

\begin{definition}[P-groups]\mml{group_10:def 17}
Let $G$ be a group. Let $p\in\NN$.
We say $G$ is a \define{$p$-Group} to mean there exists some $r\in\NN$
such that $\card{G}=p^{r}$.
\end{definition}

\begin{example}[Cyclic groups are p-groups]
Let $p\in\NN$ be nonzero.
Then the cyclic group $\CyclicGroup{p}$ is a $p$-group.
\end{example}

\begin{definition}[Sylow $p$-subgroup]\mml{group_10:def 18}
Let $p\in\NN$ be a natural number. Let $G$ be a group. Let $P\subgroup G$
be a subgroup of $G$. We define [the predicate] $P$ \define{is a Sylow $p$-subgroup}
if $P$ is a $p$-group and $p$ does not divide $\Index{G}{P}$.
\end{definition}

\begin{definition}[The set of all Sylow $p$-subgroups]\mml{group_10:def 19}
Let $G$ be a group. Let $p\in\NN$ be a natural number.
We define the term the \define{Set of all Sylow $p$-Subgroups} of $G$
to be the subset $\Syl_{p}(G)$ of $\subgroups{G}$ equal to
$\{H\subgroup G\mid H\mbox{ is Sylow $p$-subgroup of }G\}$.
\end{definition}

\begin{remark}[Sylow's theorems are never numbered consistently]
Everyone agrees there are three Sylow's theorems. No one agrees with
the ordering of those theorems. Even the presentation of the three
theorems has some variation, which is rather disappointing. I'm rather
fond of I.~Martin Isaac's naming conventions: instead of using
ordinals (``The First Sylow Theorem'', etc.), using designations
(``Sylow E'' for the existence of a Sylow subgroup, % 1.7 of FGT
``Sylow C'' for any two Sylow $p$-subgroups are \emph{conjugate} to
each other,  % 1.12 of FGT
and ``Sylow D'' for the \emph{development} result that any
$p$-subgroup is contained in a Sylow $p$-subgroup). % 1.14 of FGT
Bourbaki combines Sylow C and D into the Second Sylow Theorem
(\textit{Algebra} I~\S6.6, Theorem~3).
\end{remark}

\begin{theorem}[Existence of a Sylow subgroup]\mml{group_10:10}
Let $G$ be a finite group.
Let $p$ be a prime number.
Then there exists a strict subgroup $P\subgroup G$ such that $P$ is a
Sylow $p$-subgroup of $G$.
\end{theorem}

\begin{theorem}[Any $p$-group is contained in a Sylow subgroup]\mml{group_10:12}
Let $G$ be a finite group. Let $p$ be a prime number.
Let $H$ be a $p$-subgroup of $G$.
Then there exists a $P\in\Syl_{p}(G)$ such that $H\subgroup P$ is a
subgroup of $P$.
\end{theorem}

\begin{theorem}[Any two Sylow $p$-subgroups are conjugates of each other]\mml{group_10:12}
Let $G$ be a finite group. Let $p$ be a prime number.
For any $P_{1}\in\Syl_{p}(G)$ and $P_{2}\in\Syl_{p}(G)$, there exists
a $g\in G$ such that $P_{1}=P_{2}^{g}$.
\end{theorem}

\begin{theorem}[Number of Sylow $p$-subgroups divides order of group and is coprime to $p$]\mml{group_10:13}
Let $G$ be a finite group.
Let $p$ be a prime number.
The $\card{\Syl_{p}(G)}\equiv1\pmod{p}$, and $\card{\Syl_{p}(G)}$
divides $\card{G}$.
\end{theorem}
